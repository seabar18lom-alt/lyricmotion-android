cmake_minimum_required(VERSION 3.22)
project(lyricmotion_whisper LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
include(FetchContent)
FetchContent_Declare(whisper SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../../../whisper-src")
FetchContent_MakeAvailable(whisper)
add_library(lyricwhisper SHARED lyricwhisper.cpp)
target_link_libraries(lyricwhisper PRIVATE whisper android log)
